## Dasar Teori
*Array* merupakan kumpulan data dengan nama yang sama, di mana setiap elemennya memiliki tipe data yang sama pula. Setiap elemen *array* diakses melalui indeks, dengan elemen pertama berindeks 0 dan elemen terakhir berindeks (n-1) untuk *array* berukuran n [1]. *Array* dapat berupa satu dimensi (larik tunggal), dua dimensi (menyerupai tabel), maupun berdimensi banyak untuk kebutuhan penyimpanan data yang lebih kompleks.

Selain *array*, konsep *pointer* juga menjadi salah satu dasar penting dalam pemrograman C++. *Pointer* adalah variabel yang menyimpan alamat memori dari variabel lain, sehingga memungkinkan suatu variabel diakses secara tidak langsung melalui alamatnya [2]. Operator `&` digunakan untuk mendapatkan alamat memori suatu variabel, sedangkan operator `*` digunakan untuk mengakses nilai dari variabel yang ditunjuk oleh *pointer* tersebut. Terdapat keterhubungan yang kuat antara *array* dan *pointer*, karena nama suatu *array* pada dasarnya merupakan alamat dari elemen pertamanya.

*Fungsi* adalah blok kode yang dirancang untuk melaksanakan tugas tertentu, bertujuan membuat program lebih terstruktur dan mengurangi duplikasi kode [1]. Fungsi yang tidak mengembalikan nilai disebut *prosedur*, yang dalam C++ dikenal sebagai fungsi bertipe `void`. Dalam pemanggilan fungsi, terdapat perbedaan antara *parameter formal* (variabel pada deklarasi fungsi) dan *parameter aktual* (nilai yang digunakan saat pemanggilan fungsi).

Terdapat tiga cara melewatkan parameter ke dalam fungsi, yaitu:
1. ***Call by value*** — nilai dari parameter aktual disalin ke parameter formal, sehingga perubahan pada parameter formal tidak memengaruhi variabel aslinya.
2. ***Call by pointer*** — alamat variabel dilewatkan ke dalam fungsi menggunakan *pointer*, sehingga perubahan pada parameter formal akan memengaruhi variabel aslinya.
3. ***Call by reference*** — serupa dengan *call by pointer* dalam hal dapat mengubah nilai variabel asli, namun dengan sintaks yang lebih sederhana menggunakan operator `&` pada deklarasi parameter fungsi, tanpa memerlukan operator tambahan saat pemanggilan fungsi [2].

### 1. Buatlah program yang dapat melakukan operasi penjumlahan, pengurangan, dan perkalian matriks 3x3.

```C++
#include <iostream>
using namespace std;

const int N = 3;

void inputMatriks(int mat[N][N], string nama){
    cout << "Masukkan elemen matriks " << nama << ":\n";
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            cout << nama << "[" << i << "][" << j << "] = ";
            cin >> mat[i][j];
        }
    }
}

void tampilMatriks(int mat[N][N]){
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            cout << mat[i][j] << "\t";
        }
        cout << endl;
    }
}

int main(){
    int A[N][N], B[N][N];
    int jumlah[N][N], kurang[N][N], kali[N][N];

    inputMatriks(A, "A");
    inputMatriks(B, "B");

    // Penjumlahan dan pengurangan
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            jumlah[i][j] = A[i][j] + B[i][j];
            kurang[i][j] = A[i][j] - B[i][j];
        }
    }

    // Perkalian matriks
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            kali[i][j] = 0;
            for (int k = 0; k < N; k++) {
                kali[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    cout << "\nMatriks A:\n";
    tampilMatriks(A);
    cout << "\nMatriks B:\n";
    tampilMatriks(B);

    cout << "\nHasil Penjumlahan (A+B):\n";
    tampilMatriks(jumlah);

    cout << "\nHasil Pengurangan (A-B):\n";
    tampilMatriks(kurang);

    cout << "\nHasil Perkalian (A*B):\n";
    tampilMatriks(kali);

    return 0;
}
```

### 2. Berdasarkan guided pointer dan reference sebelumnya, buatlah keduanya dapat menukar nilai dari 3 variabel.

```C++
#include <iostream>
using namespace std;

// (A) Menukar 3 variabel menggunakan pointer
void tukarTigaPointer(int *x, int *y, int *z){
    int temp = *x;
    *x = *y;
    *y = *z;
    *z = temp;
}

// (B) Menukar 3 variabel menggunakan reference
void tukarTigaReference(int &x, int &y, int &z){
    int temp = x;
    x = y;
    y = z;
    z = temp;
}

int main(){
    int a, b, c;

    cout << "=== Menukar 3 variabel menggunakan POINTER ===\n";
    cout << "Masukkan nilai a, b, c: ";
    cin >> a >> b >> c;
    cout << "Sebelum ditukar : a=" << a << " b=" << b << " c=" << c << endl;
    tukarTigaPointer(&a, &b, &c);
    cout << "Setelah ditukar : a=" << a << " b=" << b << " c=" << c << endl;

    cout << "\n=== Menukar 3 variabel menggunakan REFERENCE ===\n";
    cout << "Masukkan nilai a, b, c: ";
    cin >> a >> b >> c;
    cout << "Sebelum ditukar : a=" << a << " b=" << b << " c=" << c << endl;
    tukarTigaReference(a, b, c);
    cout << "Setelah ditukar : a=" << a << " b=" << b << " c=" << c << endl;

    return 0;
}
```

### 3. Buatlah program yang dapat mencari nilai minimum, maksimum, dan rata-rata dari array arrA = {11, 8, 5, 7, 12, 26, 3, 54, 33, 55} menggunakan function cariMinimum(), cariMaksimum(), dan prosedur hitungRataRata(), dengan menu switch-case.

```C++
#include <iostream>
using namespace std;

const int MAX = 10;

void tampilkanArray(int arr[], int n){
    cout << "Isi array: ";
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int cariMinimum(int arr[], int n){
    int min = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }
    }
    return min;
}

int cariMaksimum(int arr[], int n){
    int maks = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > maks) {
            maks = arr[i];
        }
    }
    return maks;
}

void hitungRataRata(int arr[], int n){
    int total = 0;
    for (int i = 0; i < n; i++) {
        total += arr[i];
    }
    float rata = (float) total / n;
    cout << "Rata-rata = " << rata << endl;
}

int main(){
    int arrA[MAX] = {11, 8, 5, 7, 12, 26, 3, 54, 33, 55};
    int pilihan;

    do {
        cout << "\n--- Menu Program Array ---\n";
        cout << "1. Tampilkan isi array\n";
        cout << "2. Cari nilai maksimum\n";
        cout << "3. Cari nilai minimum\n";
        cout << "4. Hitung nilai rata-rata\n";
        cout << "5. Keluar\n";
        cout << "Pilihan Anda: ";
        cin >> pilihan;

        switch (pilihan) {
            case 1:
                tampilkanArray(arrA, MAX);
                break;
            case 2:
                cout << "Nilai maksimum = " << cariMaksimum(arrA, MAX) << endl;
                break;
            case 3:
                cout << "Nilai minimum = " << cariMinimum(arrA, MAX) << endl;
                break;
            case 4:
                hitungRataRata(arrA, MAX);
                break;
            case 5:
                cout << "Program selesai.\n";
                break;
            default:
                cout << "Pilihan tidak valid!\n";
        }
    } while (pilihan != 5);

    return 0;
}
```
